cmake_minimum_required(VERSION 3.20)
project(StellarGradientCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(STELLAR_WARNINGS_AS_ERRORS "Treat core warnings as errors" OFF)
option(STELLAR_SANITIZE "Enable Address/Undefined sanitizers" OFF)
option(STELLAR_TSAN "Enable ThreadSanitizer" OFF)

add_library(stellar_core STATIC
    src/core/RenderPlan.cpp
    src/cpu/ReferenceRenderer.cpp
)
target_include_directories(stellar_core PUBLIC src)

target_compile_options(stellar_core PRIVATE
    -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wdouble-promotion
)
if(STELLAR_WARNINGS_AS_ERRORS)
    target_compile_options(stellar_core PRIVATE -Werror)
endif()
if(STELLAR_SANITIZE AND NOT MSVC)
    target_compile_options(stellar_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(stellar_core PUBLIC -fsanitize=address,undefined)
endif()
if(STELLAR_TSAN AND NOT MSVC)
    target_compile_options(stellar_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(stellar_core PUBLIC -fsanitize=thread)
endif()

add_executable(stellar_smoke tests/core_smoke.cpp)
target_link_libraries(stellar_smoke PRIVATE stellar_core)
add_executable(stellar_edge_cases tests/core_edge_cases.cpp)
target_link_libraries(stellar_edge_cases PRIVATE stellar_core)
add_executable(stellar_bench tests/core_bench.cpp)
target_link_libraries(stellar_bench PRIVATE stellar_core)
add_executable(stellar_gpu_layout tests/gpu_layout.cpp)
target_include_directories(stellar_gpu_layout PRIVATE src)
add_executable(stellar_tile_consistency tests/tile_consistency.cpp)
target_link_libraries(stellar_tile_consistency PRIVATE stellar_core)
add_executable(stellar_quality_reference tests/quality_reference.cpp)
target_link_libraries(stellar_quality_reference PRIVATE stellar_core)
add_executable(stellar_hardening tests/hardening.cpp)
target_link_libraries(stellar_hardening PRIVATE stellar_core)
add_executable(stellar_mfr_stress tests/mfr_stress.cpp)
target_link_libraries(stellar_mfr_stress PRIVATE stellar_core)

enable_testing()
add_test(NAME stellar_smoke COMMAND stellar_smoke)
add_test(NAME stellar_edge_cases COMMAND stellar_edge_cases)
add_test(NAME stellar_gpu_layout COMMAND stellar_gpu_layout)
add_test(NAME stellar_tile_consistency COMMAND stellar_tile_consistency)
add_test(NAME stellar_quality_reference COMMAND stellar_quality_reference)
add_test(NAME stellar_hardening COMMAND stellar_hardening)
add_test(NAME stellar_mfr_stress COMMAND stellar_mfr_stress)
